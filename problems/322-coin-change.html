Java · 动态规划填表
class Solution {
    public int coinChange(int[] coins, int amount) {
        int[] dp = new int[amount + 1];
        Arrays.fill(dp, amount + 1); // 充当 ∞
        dp[0] = 0;
        for (int i = 1; i <= amount; i++) {
            for (int c : coins) {
                if (c <= i) {
                    dp[i] = Math.min(dp[i], dp[i - c] + 1);
                }
            }
        }
        return dp[amount] > amount ? -1 : dp[amount];
    }
}

Java · 记忆化搜索
class Solution {
    private int[] memo;
    public int coinChange(int[] coins, int amount) {
        memo = new int[amount + 1];
        Arrays.fill(memo, -2);       // -2 = 尚未计算
        memo[0] = 0;
        return dfs(coins, amount);
    }
    private int dfs(int[] coins, int rem) {
        if (memo[rem] != -2) return memo[rem];
        int best = Integer.MAX_VALUE;
        for (int c : coins) {
            if (c <= rem) {
                int sub = dfs(coins, rem - c);
                if (sub != -1) best = Math.min(best, sub + 1);
            }
        }
        return memo[rem] =
            best == Integer.MAX_VALUE ? -1 : best;
    }
}
Python · 记忆化搜索
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        @cache  # 放在方法内，避免跨用例共享
        def dfs(rem):
            if rem == 0:
                return 0
            best = inf
            for c in coins:
                if c <= rem:
                    best = min(best, dfs(rem - c) + 1)
            return best

        answer = dfs(amount)
        return -1 if answer == inf else answer

207. 课程表 · Kahn BFS 分批动画

只做一件事：不断完成“现在就能学”的课程，看看最后完成数 count 是否等于课程总数。
先用一句话理解每一步
① 入度
还欠几门先修课入度 2，就是还有 2 门先修课没完成。
② 入队
入度 0，现在就能学把所有入度 0 的课程放进队列。
③ 完成
队首出队，count + 1完成这门课，并删除它指向后续课程的边。
④ 解锁
后续课程入度 −1若入度降到 0，它被解锁并加入队尾。
⑤ 判断
队列空时比较数量count == numCourses 才能完成全部课程。
为什么代码有 for _ in range(len(q))？每次 while 开始就冻结本批数量。过程中刚解锁并入队的课程属于下一批，不会被当前固定次数的 for 立刻处理。
箭头始终是“先修课 → 后续课程”。1<-0 对应 LeetCode [1,0]，意思是先学 0 才能学 1，所以画边 0 → 1。课程编号顺序不重要，本题只返回能否完成。
示例： 默认：菱形 DAG 两节点环 三节点环 自环 多组件 无依赖
严格语法：每项必须是 course<-prerequisite；编号须在 0～numCourses−1，最多 36 条边，重复边会被拒绝。空输入表示无依赖；自环是合法输入，但会导致无法完成。

from collections import deque

class Solution:
    def canFinish(self, numCourses: int, prerequisites: list[list[int]]) -> bool:
        graph = [[] for _ in range(numCourses)]
        in_degree = [0] * numCourses
        for course, prerequisite in prerequisites:
            graph[prerequisite].append(course)
            in_degree[course] += 1

        q = deque(i for i in range(numCourses) if in_degree[i] == 0)
        count = 0
        while q:
            # 本轮只处理 while 开始时已经在队列里的课程
            for _ in range(len(q)):
                idx = q.popleft()
                count += 1
                for item in graph[idx]:
                    in_degree[item] -= 1
                    if in_degree[item] == 0:
                        q.append(item)  # 留给下一批
        return count == numCourses
class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        List<List<Integer>> graph = new ArrayList<>();
        int[] inDegree = new int[numCourses];
        for (int i = 0; i < numCourses; i++) graph.add(new ArrayList<>());
        for (int[] edge : prerequisites) {
            graph.get(edge[1]).add(edge[0]);
            inDegree[edge[0]]++;
        }

        Queue<Integer> q = new ArrayDeque<>();
        for (int i = 0; i < numCourses; i++)
            if (inDegree[i] == 0) q.offer(i);
        int count = 0;
        while (!q.isEmpty()) {
            int batchSize = q.size(); // 冻结本批次数
            for (int i = 0; i < batchSize; i++) {
                int idx = q.poll();
                count++;
                for (int item : graph.get(idx)) {
                    inDegree[item]--;
                    if (inDegree[item] == 0) q.offer(item);
                }
            }
        }
        return count == numCourses;
    }
}

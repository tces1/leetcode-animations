5. 最长回文子串 · 二维 DP / 中心扩展

① 定义
dp[i][j] = 子串 s[i..j] 是否回文。
② 递推
两端 s[i]==s[j]，且里面 dp[i+1][j-1] 也回文。
③ 填表顺序
必须按长度从小到大填，才能先得到更靠里的格子。
④ 边界
长度 1 一定回文；长度 2 只要两端相等。
① 枚举中心
每个字符做奇数中心，每条缝隙做偶数中心，共 2n-1 个。
② 双向扩展
L 左移、R 右移，只要 s[L]==s[R] 就继续。
③ 碰壁即停
越界或两端不等，这个中心的扩展结束。
④ 刷新最长
每次扩成功都和当前答案比长度，更长就替换。
回文由中心唯一确定。奇数回文中心是一个字符（如 aba 的 b）；偶数回文中心是两个字符之间的缝隙（如 abba 的两个 b 中间）。时间仍是 O(n²)，但只要几个指针，空间 O(1)。
① 二维 DP
时间 O(n²) · 空间 O(n²)
按长度填上三角，依赖内格

# Python · 二维 DP(区间) · 时间/空间 O(n²)
class Solution:
    def longestPalindrome(self, s: str) -> str:
        n = len(s)
        dp = [[False] * n for _ in range(n)]
        start, maxLen = 0, 1
        for i in range(n):
            dp[i][i] = True                 # 长度 1
        for L in range(2, n + 1):           # 按长度从小到大
            for i in range(n - L + 1):
                j = i + L - 1
                if s[i] != s[j]:
                    dp[i][j] = False
                elif L == 2:
                    dp[i][j] = True         # 两端相等且长度 2
                else:
                    dp[i][j] = dp[i + 1][j - 1]   # 看里面那格
                if dp[i][j] and L > maxLen:
                    start, maxLen = i, L
        return s[start:start + maxLen]
// Java · 二维 DP(区间) · 时间/空间 O(n²)
class Solution {
    public String longestPalindrome(String s) {
        int n = s.length();
        boolean[][] dp = new boolean[n][n];
        int start = 0, maxLen = 1;
        for (int i = 0; i < n; i++) dp[i][i] = true;   // 长度 1
        for (int L = 2; L <= n; L++) {                  // 按长度从小到大
            for (int i = 0; i + L - 1 < n; i++) {
                int j = i + L - 1;
                if (s.charAt(i) != s.charAt(j)) dp[i][j] = false;
                else if (L == 2) dp[i][j] = true;         // 两端相等且长度 2
                else dp[i][j] = dp[i + 1][j - 1];         // 看里面那格
                if (dp[i][j] && L > maxLen) { start = i; maxLen = L; }
            }
        }
        return s.substring(start, start + maxLen);
    }
}
# Python · 中心扩展 · 时间 O(n²) 空间 O(1)
class Solution:
    def longestPalindrome(self, s: str) -> str:
        n = len(s)
        resp = ""
        for mid in range(n):
            # 奇数：以 s[mid] 为中心
            dl = dr = mid
            while 0 <= dl and dr < n:
                if s[dl] != s[dr]:
                    break
                if dr - dl + 1 > len(resp):
                    resp = s[dl:dr + 1]
                dl -= 1
                dr += 1
            # 偶数：以 s[mid] 与 s[mid+1] 的缝隙为中心
            dl, dr = mid, mid + 1
            while 0 <= dl and dr < n:
                if s[dl] != s[dr]:
                    break
                if dr - dl + 1 > len(resp):
                    resp = s[dl:dr + 1]
                dl -= 1
                dr += 1
        return resp
// Java · 中心扩展 · 时间 O(n²) 空间 O(1)
class Solution {
    public String longestPalindrome(String s) {
        int n = s.length();
        String resp = "";
        for (int mid = 0; mid < n; mid++) {
            // 奇数：以 s[mid] 为中心
            int dl = mid, dr = mid;
            while (dl >= 0 && dr < n) {
                if (s.charAt(dl) != s.charAt(dr)) break;
                if (dr - dl + 1 > resp.length()) {
                    resp = s.substring(dl, dr + 1);
                }
                dl--; dr++;
            }
            // 偶数：以缝隙为中心
            dl = mid; dr = mid + 1;
            while (dl >= 0 && dr < n) {
                if (s.charAt(dl) != s.charAt(dr)) break;
                if (dr - dl + 1 > resp.length()) {
                    resp = s.substring(dl, dr + 1);
                }
                dl--; dr++;
            }
        }
        return resp;
    }
}

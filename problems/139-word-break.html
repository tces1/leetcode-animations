Java · 一维 DP
class Solution {
    public boolean wordBreak(String s, List<String> wordDict) {
        Set<String> set = new HashSet<>(wordDict);
        int n = s.length();
        boolean[] dp = new boolean[n + 1];
        dp[0] = true;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < i; j++) {
                if (dp[j] && set.contains(s.substring(j, i))) {
                    dp[i] = true;
                    break;
                }
            }
        }
        return dp[n];
    }
}

class Solution:
    def wordBreak(self, s: str, wordDict: List[str]) -> bool:
        ws = set(wordDict)
        n = len(s)
        dp = [False] * (n + 1)
        dp[0] = True
        for i in range(1, n + 1):
            for j in range(i):
                if dp[j] and s[j:i] in ws:
                    dp[i] = True
                    break
        return dp[n]
Java · 记忆化 DFS
class Solution {
    public boolean wordBreak(String s, List<String> wordDict) {
        int n = s.length();
        Boolean[] memo = new Boolean[n + 1];
        return dfs(0, s, wordDict, memo);
    }
    private boolean dfs(int start, String s, List<String> dict, Boolean[] memo) {
        if (start == s.length()) return true;
        if (memo[start] != null) return memo[start];
        for (String w : dict) {
            int wn = w.length();
            if (s.startsWith(w, start) && dfs(start + wn, s, dict, memo)) {
                return memo[start] = true;
            }
        }
        return memo[start] = false;
    }
}
Python · 记忆化 DFS
class Solution:
    def wordBreak(self, s: str, wordDict: List[str]) -> bool:
        n = len(s)

        @cache
        def sub(start):
            if start == n:
                return True
            for item in wordDict:
                wn = len(item)
                if s[start:start + wn] == item and sub(start + wn):
                    return True
            return False

        return sub(0)
